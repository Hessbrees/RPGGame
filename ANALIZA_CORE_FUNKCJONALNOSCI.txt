        print(f"TURA {turn + 1}")
        print('='*60)
        
        # Sprawdź czy gra się skończyła
        if orchestrator.is_game_over():
            print("\n=== KONIEC GRY ===")
            print(orchestrator.get_ending())
            break
        
        # GM decyduje o trybie konwersacji
        mode = orchestrator.game_state.conversation_mode
        print(f"[Tryb: {mode}]\n")
        
        if mode == ConversationMode.FREE:
            # ROZMOWA SWOBODNA - gracze AI rozmawiają
            orchestrator.run_free_conversation()
            
        elif mode == ConversationMode.TURN_BASED:
            # ROZMOWA TUROWA - po kolei
            orchestrator.run_turn_based_round()
        
        # GM analizuje czy zmienić tryb
        orchestrator.game_master.evaluate_and_switch_mode()
        
        # Wyświetl aktualny stan
        orchestrator.display_game_state()
        
        time.sleep(2)  # Krótka pauza dla czytelności
    
    print("\n=== KONIEC SYMULACJI ===")

class GameOrchestrator:
    def __init__(self, llm, embed_model, vibe="zrównoważona", num_players=2):
        # RAG-i (bez zmian)
        self.lore_rag = LoreRAG(llm, embed_model)
        self.npc_rag = NpcRAG(llm, embed_model)
        self.judge_rag = JudgeRAG(llm, embed_model)
        self.chronicle_rag = ChronicleRAG(llm, embed_model)
        self.reranker = Reranker(top_k=3)
        
        # GM
        self.game_master = GameMasterAgent(llm, vibe)
        
        # GameState
        self.game_state = GameState(vibe=GameVibe(vibe))
        
        # ✨ NOWE: Gracze AI
        self.players: list[PlayerAgent] = []
        self._initialize_players(llm, num_players)
        
        # ✨ NOWE: Wspólna szyna komunikacyjna
        self.message_history: list[dict] = []
        
        # ✨ NOWE: Indeks aktywnego gracza (dla TURN_BASED)
        self.active_player_index = 0
    
    def _initialize_players(self, llm, num_players):
        """Tworzy graczy AI z predefiniowanymi personami."""
        personas = [
            {
                "name": "Arin",
                "class": "złodziej",
                "traits": "przebiegły, skryty, nie ufa obcym",
                "backstory": "Wychował się na ulicach wielkiego miasta"
            },
            {
                "name": "Theron",
                "class": "wojownik",
                "traits": "odważny, honorowy, lojalny",
                "backstory": "Były żołnierz Zakonu Srebrnego Ostrza"
            }
        ]
        
        for i in range(min(num_players, len(personas))):
            persona = personas[i]
            player_agent = PlayerAgent(llm, persona)
            self.players.append(player_agent)
            
            # Dodaj do GameState
            player_state = PlayerState(
                name=persona["name"],
                persona=persona,
                hp=100,
                inventory=["miecz krótki", "pochodnia", "prowiant"],
                active=True
            )
            self.game_state.players.append(player_state)
    
    def start_adventure(self) -> str:
        """GM rozpoczyna przygodę."""
        intro = self.game_master.narrate(
            situation="Początek przygody w Lesie Szeptów",
            context=self.lore_rag.query("Las Szeptów")
        )
        self.broadcast_message({
            "turn": 0,
            "speaker": "GM",
            "type": "narration",
            "content": intro,
            "visible_to": ["all"]
        })
        return intro
    
    def run_free_conversation(self):
        """Rozmowa swobodna - gracze AI rozmawiają między sobą."""
        max_exchanges = 5
        
        for i in range(max_exchanges):
            # Każdy gracz po kolei (ale bez sztywnej tury bojowej)
            for player_agent in self.players:
                # Gracz widzi co mówili inni
                context = self.get_visible_history(player_agent.persona["name"])
                
                # Gracz AI decyduje co powiedzieć/zrobić
                action = player_agent.decide_action(
                    situation=context[-500:],  # ostatnie 500 znaków
                    available_actions=None
                )
                
                print(f"{player_agent.persona['name']}: {action}")
                
                # Broadcast do wszystkich
                self.broadcast_message({
                    "turn": self.game_state.current_turn,
                    "speaker": player_agent.persona["name"],
                    "type": "dialog",
                    "content": action,
                    "visible_to": ["all"]
                })
                
                # Zapisz w kronice
                self.chronicle_rag.add_entry(
                    f"[Tura {self.game_state.current_turn}] "
                    f"{player_agent.persona['name']}: {action}"
                )
            
            # GM sprawdza czy zakończyć rozmowę
            if self.game_master.should_end_free_conversation(self.message_history):
                print("\nGM: (kończy rozmowę)")
                self.game_state.conversation_mode = ConversationMode.TURN_BASED
                break
    
    def run_turn_based_round(self):
        """Jedna runda turowa - każdy gracz po kolei."""
        for player_state in self.game_state.players:
            if not player_state.active:
                continue  # Gracz nieprzytomny/martwy
            
            # Znajdź odpowiadającego PlayerAgent
            player_agent = next(
                (p for p in self.players if p.persona["name"] == player_state.name),
                None
            )
            
            if not player_agent:
                continue
            
            # Gracz AI podejmuje decyzję
            context = self.get_visible_history(player_state.name)
            action = player_agent.decide_action(
                situation=context[-500:],
                available_actions=["Atak", "Obrona", "Użyj przedmiotu", "Uciekaj"]
            )
            
            print(f"\n{player_state.name}: {action}")
            
            # GM przetwarza akcję
            result = self.process_player_action(action, player_state.name)
            print(f"GM: {result}")
            
            # Broadcast
            self.broadcast_message({
                "turn": self.game_state.current_turn,
                "speaker": player_state.name,
                "type": "action",
                "content": action,
                "visible_to": ["all"]
            })
        
        # Następna tura
        self.game_state.next_turn()
    
    def broadcast_message(self, message: dict):
        """Dodaje wiadomość do wspólnej szyny komunikacyjnej."""
        self.message_history.append(message)
    
    def get_visible_history(self, player_name: str) -> str:
        """Zwraca historię wiadomości widoczną dla gracza."""
        visible = [
            f"{msg['speaker']}: {msg['content']}"
            for msg in self.message_history
            if "all" in msg["visible_to"] or player_name in msg["visible_to"]
        ]
        return "\n".join(visible[-10:])  # ostatnie 10 wiadomości
    
    def display_game_state(self):
        """Wyświetla aktualny stan gry."""
        print("\n--- STAN GRY ---")
        for player in self.game_state.players:
            status = "✅" if player.active else "❌"
            print(f"{status} {player.name}: HP {player.hp}/100")
        print(f"Tura: {self.game_state.current_turn}")
        print(f"Trudność: {self.game_state.difficulty}")
